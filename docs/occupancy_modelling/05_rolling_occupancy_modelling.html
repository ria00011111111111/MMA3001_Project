occupancy_modelling.05_rolling_occupancy_modelling

Evaluate occupancy forecasting using expanding chronological windows.

The script aggregates occupancy sensor readings into five-minute intervals, constructs features using current and previous observations, and predicts the occupancy state in the exact next five-minute interval.

A Random Forest classifier is evaluated against two simple baselines: a majority-class baseline and a persistence baseline. Evaluation uses three expanding chronological windows, with training targets preceding the corresponding test period.

The script saves evaluation metrics, individual predictions, and forecast row counts to a dedicated output directory. It does not overwrite the main forecast pipeline's outputs.

Important assumptions: - The latest observation in each floor-space/time bin represents that bin. - A target exists only when the exact next five-minute bin is present. - Features are calculated using current and earlier observations only. - Windows are split by target timestamp, ensuring training targets precede test targets. - The persistence baseline predicts that the current state will continue into the next interval. - The majority baseline always predicts the most common training class.

Outputs: - rolling_evaluation_metrics.csv: Evaluation metrics for each model and chronological window. - rolling_evaluation_predictions.csv: Actual and predicted occupancy states for each evaluated row. - rolling_evaluation_row_counts.csv: Counts describing target matching and forecast-row eligibility.

Notes: This script evaluates recorded occupancy sensor states rather than independently verified physical occupancy. A nominal five-minute target interval does not necessarily represent exactly five minutes of elapsed time between the original sensor observations.

114def load_occupancy_data(zip_path: Path) -> pd.DataFrame:
115    """Load and aggregate valid occupancy observations from a ZIP archive.
116
117    Reads the source CSV in chunks to limit the amount of data loaded at once.
118    Invalid rows with missing floor-space identifiers, occupancy states, or
119    timestamps are removed. Timestamps are converted to UTC and assigned to
120    five-minute intervals.
121
122    When multiple observations exist for the same floor space and interval,
123    only the latest source observation is retained.
124
125    Args:
126        zip_path: Path to the ZIP archive containing the occupancy CSV.
127
128    Returns:
129        A DataFrame containing one observation per floor space and
130        five-minute interval, sorted chronologically within each floor space.
131        The DataFrame includes the original observation timestamp and the
132        corresponding interval timestamp.
133
134    Raises:
135        FileNotFoundError: If the specified ZIP archive does not exist.
136        ValueError: If the archive does not contain exactly one CSV file,
137            contains no data rows, or contains no valid observations.
138
139    Note:
140        The interval timestamp is calculated by flooring the original
141        observation timestamp to a five-minute boundary. It is not a
142        replacement for the original sensor timestamp.
143    """
144    # Check that the source dataset is available before opening the archive.
145    if not zip_path.is_file():
146        raise FileNotFoundError(f"Input ZIP file not found: {zip_path}")
147
148    chunks = []
149    rows_read = 0
150
151    with ZipFile(zip_path) as archive:
152        # Require a single source CSV to avoid accidentally combining
153        # unrelated datasets from the same archive.
154        csv_files = [
155            name for name in archive.namelist()
156            if name.lower().endswith(".csv")
157        ]
158        if len(csv_files) != 1:
159            raise ValueError(
160                "Expected exactly one CSV in the ZIP archive, "
161                f"found {len(csv_files)}."
162            )
163
164        with archive.open(csv_files[0]) as source:
165            # Read only the fields required for aggregation and forecasting.
166            for chunk in pd.read_csv(
167                source,
168                usecols=["floorspaceid", TARGET, "collecteddate"],
169                chunksize=CHUNK_SIZE,
170                low_memory=False,
171            ):
172                rows_read += len(chunk)
173
174                # Convert timestamps to a consistent timezone and treat
175                # unparseable timestamps as missing rather than valid data.
176                chunk["collecteddate"] = pd.to_datetime(
177                    chunk["collecteddate"], utc=True, errors="coerce"
178                )
179
180                # Remove records that cannot be assigned to a floor space,
181                # occupancy state, or time interval.
182                chunk = chunk.dropna(
183                    subset=["floorspaceid", TARGET, "collecteddate"]
184                )
185
186                # Assign each observation to its corresponding five-minute bin.
187                chunk["interval"] = chunk["collecteddate"].dt.floor("5min")
188                chunks.append(chunk)
189                print(f"Read {rows_read:,} raw rows...")
190
191    if not chunks:
192        raise ValueError("The source CSV contained no data rows.")
193
194    # Combine the cleaned chunks before deduplication so duplicate
195    # floor-space/interval records across chunk boundaries are also handled.
196    data = pd.concat(chunks, ignore_index=True)
197    del chunks
198
199    if data.empty:
200        raise ValueError("No valid occupancy rows remained after cleaning.")
201
202    # Retain the latest recorded observation when multiple readings fall
203    # within the same floor-space/time bin.
204    data = data.sort_values(
205        ["floorspaceid", "interval", "collecteddate"]
206    ).drop_duplicates(
207        subset=["floorspaceid", "interval"], keep="last"
208    )
209
210    # Sort by floor space and time so subsequent grouped operations can
211    # reliably identify previous observations.
212    return data.sort_values(
213        ["floorspaceid", "interval"]
214    ).reset_index(drop=True)

217def build_forecast_rows(data: pd.DataFrame) -> tuple[pd.DataFrame, dict]:
218    """Construct forecasting features and match exact next-interval targets.
219
220    Each aggregated observation becomes a candidate forecast origin. Features
221    are constructed from the current observation, the previous observation
222    for the same floor space, and the timestamp of the current interval.
223
224    The target is obtained by matching the current row to an observation
225    belonging to the exact next five-minute interval for the same floor
226    space. Rows without a matching target, previous state, or valid
227    observation-gap feature are excluded from the eligible dataset.
228
229    Args:
230        data: Aggregated occupancy observations containing floor-space IDs,
231            occupancy states, original timestamps, and interval timestamps.
232
233    Returns:
234        A tuple containing:
235
236        - A DataFrame of eligible forecast rows, including the engineered
237          features and the matched future occupancy state in ``TARGET``.
238        - A dictionary of row counts describing the target-matching and
239          eligibility process.
240
241    Note:
242        The target is matched using the interval timestamp, not by selecting
243        the next available observation. Consequently, an observation is
244        excluded if the exact next five-minute bin is absent.
245
246        The observation-gap feature uses the original ``collecteddate``
247        timestamps. It is therefore possible for the elapsed time between
248        sensor observations to differ from five minutes, even when their
249        interval timestamps are consecutive.
250    """
251    # Work on a copy so feature engineering does not modify the aggregated
252    # observations used elsewhere in the evaluation.
253    data = data.copy()
254    data["current_state"] = data[TARGET]
255
256    # Calculate previous observations independently within each floor space
257    # to prevent readings from different spaces being mixed together.
258    grouped = data.groupby("floorspaceid", sort=False)
259    data["previous_state"] = grouped[TARGET].shift(1)
260    data["previous_timestamp"] = grouped["collecteddate"].shift(1)
261
262    # Measure the elapsed time between original sensor readings rather than
263    # assuming that consecutive interval bins are exactly five minutes apart.
264    data["gap_minutes"] = (
265        data["collecteddate"] - data["previous_timestamp"]
266    ).dt.total_seconds() / 60
267
268    # Exclude non-positive gaps and cap unusually large gaps before applying
269    # the logarithmic transform to reduce the influence of extreme values.
270    data.loc[data["gap_minutes"] <= 0, "gap_minutes"] = np.nan
271    data["log_gap_minutes"] = np.log1p(
272        data["gap_minutes"].clip(upper=1440)
273    )
274
275    # Extract calendar features from the current forecast-origin interval.
276    data["hour"] = data["interval"].dt.hour
277    data["day_of_week"] = data["interval"].dt.dayofweek
278    data["month"] = data["interval"].dt.month
279    data["is_weekend"] = (data["day_of_week"] >= 5).astype(int)
280
281    # Define the exact future interval that the current observation must predict.
282    data["target_interval"] = (
283        data["interval"] + pd.Timedelta(minutes=FORECAST_MINUTES)
284    )
285
286    # Build a lookup table of observed occupancy states indexed by their
287    # floor space and interval, allowing exact future-target matching.
288    future = data[["floorspaceid", "interval", TARGET]].rename(
289        columns={
290            "interval": "target_interval",
291            TARGET: "future_state",
292        }
293    )
294
295    # Match each forecast origin to the exact next interval.
296    # A left join preserves unmatched origins so they can be counted.
297    candidates = data.merge(
298        future,
299        on=["floorspaceid", "target_interval"],
300        how="left",
301        validate="one_to_one",
302        indicator=True,
303    )
304    total_rows = len(candidates)
305    missing_target_rows = int((candidates["_merge"] == "left_only").sum())
306
307    # Keep only rows with a matched future state and the historical
308    # information needed to calculate the forecasting features.
309    eligible = candidates.drop(columns="_merge").dropna(
310        subset=["future_state", "previous_state", "log_gap_minutes"]
311    ).copy()
312
313    # Store the matched future state as the prediction target and order
314    # the resulting records by target time and floor space.
315    eligible[TARGET] = eligible["future_state"]
316    eligible = eligible.sort_values(
317        ["target_interval", "floorspaceid"]
318    ).reset_index(drop=True)
319
320    # Track how many records are lost at each stage so forecast-row
321    # eligibility can be assessed independently of model performance.
322    counts = {
323        "aggregated_intervals": total_rows,
324        "rows_without_exact_next_bin": missing_target_rows,
325        "rows_with_exact_next_bin": total_rows - missing_target_rows,
326        "eligible_forecast_rows": len(eligible),
327        "rows_excluded_after_target_matching": (
328            total_rows - len(eligible)
329        ),
330    }
331    return eligible, counts

334def make_model() -> Pipeline:
335    """Create the preprocessing and Random Forest classification pipeline.
336
337    Categorical features are imputed using their most frequent values and
338    one-hot encoded. Numeric features are imputed using their median values.
339    The transformed features are then passed to a Random Forest classifier.
340
341    Returns:
342        A scikit-learn Pipeline containing the preprocessing transformer
343        and configured Random Forest classifier.
344
345    Note:
346        The one-hot encoder ignores previously unseen categories during
347        prediction. Imputation and encoding are fitted as part of the
348        pipeline, so their parameters are learned from the training data
349        when ``fit`` is called.
350    """
351    # Apply separate preprocessing to categorical and numerical inputs
352    # while keeping both transformations within a single model pipeline.
353    preprocessing = ColumnTransformer(
354        transformers=[
355            (
356                "categorical",
357                Pipeline(
358                    steps=[
359                        ("imputer", SimpleImputer(strategy="most_frequent")),
360                        ("encoder", OneHotEncoder(handle_unknown="ignore")),
361                    ]
362                ),
363                CATEGORICAL_FEATURES,
364            ),
365            (
366                "numeric",
367                Pipeline(
368                    steps=[("imputer", SimpleImputer(strategy="median"))]
369                ),
370                NUMERIC_FEATURES,
371            ),
372        ],
373        remainder="drop",
374    )
375
376    # Keeping preprocessing and classification together ensures that
377    # imputation and encoding are fitted using training data only.
378    return Pipeline(
379        steps=[
380            ("preprocessing", preprocessing),
381            (
382                "classifier",
383                RandomForestClassifier(**RANDOM_FOREST_SETTINGS),
384            ),
385        ]
386    )

389def create_windows(data: pd.DataFrame) -> list[dict]:
390    """Generate three expanding chronological evaluation windows.
391
392    Unique target timestamps are sorted chronologically and divided using
393    cutoffs at 60%, 70%, 80%, and 90% of the available timestamps.
394
395    Each window uses all eligible observations up to the preceding cutoff
396    for training, followed by the next chronological segment for testing.
397    This produces three expanding training sets and three non-overlapping
398    test periods.
399
400    Args:
401        data: Eligible forecast rows containing a ``target_interval``
402            column.
403
404    Returns:
405        A list of three dictionaries. Each dictionary contains the window
406        name and its training-end, test-start, and test-end timestamps.
407
408    Raises:
409        ValueError: If fewer than 100 unique target timestamps are available
410            or if the cutoffs cannot produce non-empty chronological windows.
411
412    Note:
413        Approximately the first 60% of unique target timestamps are used
414        for the first training set. The three test periods cover approximately
415        60–70%, 70–80%, and 80–90% of the timestamps. The final 10% is not
416        included in these evaluation windows.
417    """
418    # Split using unique target timestamps rather than individual rows so
419    # observations sharing a target time cannot be divided across sets.
420    unique_times = np.sort(data["target_interval"].dropna().unique())
421    if len(unique_times) < 100:
422        raise ValueError(
423            "Not enough unique target timestamps for rolling evaluation."
424        )
425
426    n_times = len(unique_times)
427
428    # Establish chronological boundaries for three consecutive test periods.
429    # Each later training set includes the earlier training and test periods.
430    cutoffs = [int(n_times * fraction) for fraction in (0.60, 0.70, 0.80, 0.90)]
431    if any(cutoffs[i] >= cutoffs[i + 1] for i in range(len(cutoffs) - 1)):
432        raise ValueError("Not enough timestamps to create non-empty windows.")
433
434    # Use the preceding timestamp as the training boundary and the next
435    # segment as the test period for each expanding window.
436    return [
437        {
438            "window": f"Window {index + 1}",
439            "train_end": unique_times[cutoffs[index] - 1],
440            "test_start": unique_times[cutoffs[index]],
441            "test_end": unique_times[cutoffs[index + 1] - 1],
442        }
443        for index in range(3)
444    ]

447def evaluate_windows(
448    data: pd.DataFrame,
449    windows: list[dict],
450) -> tuple[pd.DataFrame, pd.DataFrame]:
451    """Train and evaluate forecasting models across chronological windows.
452
453    For each window, the function selects training and test rows according
454    to their target timestamps. It then evaluates three prediction
455    approaches:
456
457    - Majority baseline: predicts the most common training target class.
458    - Persistence baseline: predicts that the current state continues
459      into the target interval.
460    - Random Forest: fits the configured classification pipeline using
461      the training data and predicts the test data.
462
463    Accuracy, balanced accuracy, macro F1, and weighted F1 are calculated
464    for each model and window. Individual predictions are also retained
465    for later inspection.
466
467    Args:
468        data: Eligible forecast rows containing the engineered features,
469            target labels, floor-space IDs, and target timestamps.
470        windows: Chronological window definitions generated by
471            ``create_windows``.
472
473    Returns:
474        A tuple containing:
475
476        - A DataFrame of evaluation metrics, with one row per model
477          and completed window.
478        - A DataFrame containing the actual and predicted states for
479          every evaluated test row.
480
481    Raises:
482        ValueError: If chronological overlap is detected between a
483            training and test set.
484        RuntimeError: If no evaluation windows complete successfully.
485
486    Note:
487        Windows with empty training or test sets, or fewer than two
488        training classes, are skipped. Training data expands between
489        windows, while the test periods remain separate.
490
491        The F1 metrics use ``STATE_ORDER`` as the fixed set of class
492        labels. This ensures that the reported macro and weighted F1
493        scores use a consistent class ordering across windows.
494    """
495    results = []
496    prediction_tables = []
497
498    for window in windows:
499        name = window["window"]
500        train_end = pd.Timestamp(window["train_end"])
501        test_start = pd.Timestamp(window["test_start"])
502        test_end = pd.Timestamp(window["test_end"])
503
504        # Select all targets up to the training cutoff and only the
505        # corresponding later chronological segment for testing.
506        train = data[data["target_interval"] <= train_end].copy()
507        test = data[
508            (data["target_interval"] >= test_start)
509            & (data["target_interval"] <= test_end)
510        ].copy()
511
512        # Skip windows that cannot support a meaningful model evaluation.
513        if train.empty or test.empty:
514            print(f"Skipping {name}: empty training or test set.")
515            continue
516        if train["target_interval"].max() >= test["target_interval"].min():
517            raise ValueError(f"Chronological overlap detected in {name}.")
518        if train[TARGET].nunique() < 2:
519            print(f"Skipping {name}: fewer than two training classes.")
520            continue
521
522        # Separate the model inputs from the future occupancy state to be
523        # predicted for each training and test observation.
524        X_train, y_train = train[FEATURES], train[TARGET]
525        X_test, y_test = test[FEATURES], test[TARGET]
526
527        # The majority baseline uses only the most frequent training class.
528        majority_class = y_train.value_counts().idxmax()
529        majority_predictions = np.repeat(majority_class, len(test))
530
531        # The persistence baseline assumes the current occupancy state
532        # remains unchanged over the forecast horizon.
533        persistence_predictions = X_test["current_state"].to_numpy()
534
535        # Fit a fresh Random Forest for each window so that each model
536        # uses only the training data available at that point in time.
537        model = make_model()
538        model.fit(X_train, y_train)
539        rf_predictions = model.predict(X_test)
540
541        # Keep all three prediction approaches together for consistent
542        # metric calculation and output generation.
543        predictions_by_model = {
544            "Majority baseline": majority_predictions,
545            "Persistence baseline": persistence_predictions,
546            "Random Forest": rf_predictions,
547        }
548
549        print("\n" + "-" * 70)
550        print(name)
551        print(f"Training targets through: {train_end}")
552        print(f"Testing targets: {test_start} to {test_end}")
553        print(f"Training rows: {len(train):,}")
554        print(f"Testing rows:  {len(test):,}")
555
556        # Evaluate every model against the same actual test labels.
557        for model_name, predictions in predictions_by_model.items():
558            results.append({
559                "window": name,
560                "model": model_name,
561                "train_rows": len(train),
562                "test_rows": len(test),
563                "train_end": train_end,
564                "test_start": test_start,
565                "test_end": test_end,
566                "accuracy": accuracy_score(y_test, predictions),
567                "balanced_accuracy": balanced_accuracy_score(y_test, predictions),
568                "macro_f1": f1_score(
569                    y_test, predictions, labels=STATE_ORDER,
570                    average="macro", zero_division=0,
571                ),
572                "weighted_f1": f1_score(
573                    y_test, predictions, labels=STATE_ORDER,
574                    average="weighted", zero_division=0,
575                ),
576            })
577
578        # Preserve row-level predictions so model errors and individual
579        # occupancy-state changes can be inspected after evaluation.
580        prediction_tables.append(pd.DataFrame({
581            "window": name,
582            "target_interval": test["target_interval"].to_numpy(),
583            "floorspaceid": test["floorspaceid"].to_numpy(),
584            "actual_state": y_test.to_numpy(),
585            "majority_prediction": majority_predictions,
586            "persistence_prediction": persistence_predictions,
587            "random_forest_prediction": rf_predictions,
588        }))
589
590    # Prevent empty output files from being generated if every window
591    # was skipped during the evaluation process.
592    if not results or not prediction_tables:
593        raise RuntimeError("No rolling evaluation windows completed.")
594
595    return (
596        pd.DataFrame(results),
597        pd.concat(prediction_tables, ignore_index=True),
598    )

601def main() -> None:
602    """Run the rolling evaluation and save all generated outputs.
603
604    Loads and aggregates the occupancy sensor data, constructs eligible
605    forecasting rows, creates chronological evaluation windows, and
606    evaluates the majority baseline, persistence baseline, and Random
607    Forest classifier.
608
609    The function saves the evaluation metrics, individual predictions,
610    and row-count summary to ``OUTPUT_DIR`` before printing a summary
611    of the completed evaluation.
612
613    Raises:
614        FileNotFoundError: If the configured input ZIP file is missing.
615        ValueError: If the input data is invalid, no eligible forecast
616            rows are generated, or chronological windows cannot be created.
617        RuntimeError: If no evaluation windows complete successfully.
618    """
619    print("=" * 70)
620    print("ROLLING CHRONOLOGICAL OCCUPANCY FORECAST EVALUATION")
621    print("=" * 70)
622
623    # Load and aggregate the source sensor observations before constructing
624    # the forecast dataset.
625    data = load_occupancy_data(ZIP_FILE)
626    print(f"\nAggregated intervals: {len(data):,}")
627
628    # Match exact next-interval targets and record how many observations
629    # remain eligible for forecasting after feature construction.
630    forecast_rows, counts = build_forecast_rows(data)
631    print("\nFORECAST ROW COUNTS")
632    for label, value in counts.items():
633        print(f"{label.replace('_', ' ').capitalize()}: {value:,}")
634
635    # Stop if the source data cannot produce any valid forecasting examples.
636    if forecast_rows.empty:
637        raise ValueError("No eligible forecasting rows were generated.")
638
639    print(
640        "\nTarget period:",
641        forecast_rows["target_interval"].min(),
642        "to",
643        forecast_rows["target_interval"].max(),
644    )
645
646    # Generate the expanding chronological windows and evaluate all models
647    # using the same window boundaries for a fair comparison.
648    windows = create_windows(forecast_rows)
649    results, predictions = evaluate_windows(forecast_rows, windows)
650
651    # Save outputs in a dedicated directory to keep the rolling evaluation
652    # separate from the main occupancy forecasting pipeline.
653    OUTPUT_DIR.mkdir(parents=True, exist_ok=True)

117    """Load and aggregate valid occupancy observations from a ZIP archive.
118
119    Reads the source CSV in chunks to limit the amount of data loaded at once.
120    Invalid rows with missing floor-space identifiers, occupancy states, or
121    timestamps are removed. Timestamps are converted to UTC and assigned to
122    five-minute intervals.
123
124    When multiple observations exist for the same floor space and interval,
125    only the latest source observation is retained.
126
127    Args:
128        zip_path: Path to the ZIP archive containing the occupancy CSV.
129
130    Returns:
131        A DataFrame containing one observation per floor space and
132        five-minute interval, sorted chronologically within each floor space.

134        corresponding interval timestamp.
135
136    Raises:
137        FileNotFoundError: If the specified ZIP archive does not exist.

144        replacement for the original sensor timestamp.

147    if not zip_path.is_file():

152
153    with ZipFile(zip_path) as archive:
154        # Require a single source CSV to avoid accidentally combining
155        # unrelated datasets from the same archive.
156        csv_files = [
157            name for name in archive.namelist()
158            if name.lower().endswith(".csv")
159        ]

161            raise ValueError(

163                f"found {len(csv_files)}."
164            )

168            for chunk in pd.read_csv(

172                low_memory=False,

175
176                # Convert timestamps to a consistent timezone and treat
177                # unparseable timestamps as missing rather than valid data.
178                chunk["collecteddate"] = pd.to_datetime(
179                    chunk["collecteddate"], utc=True, errors="coerce"
180                )
181
182                # Remove records that cannot be assigned to a floor space,
183                # occupancy state, or time interval.
184                chunk = chunk.dropna(
185                    subset=["floorspaceid", TARGET, "collecteddate"]
186                )

def build_forecast_rows(data: pandas.DataFrame) -> tuple[pandas.DataFrame, dict]: 
219def build_forecast_rows(data: pd.DataFrame) -> tuple[pd.DataFrame, dict]:
220    """Construct forecasting features and match exact next-interval targets.
221
222    Each aggregated observation becomes a candidate forecast origin. Features
223    are constructed from the current observation, the previous observation
224    for the same floor space, and the timestamp of the current interval.
225
226    The target is obtained by matching the current row to an observation
227    belonging to the exact next five-minute interval for the same floor
228    space. Rows without a matching target, previous state, or valid
229    observation-gap feature are excluded from the eligible dataset.
230
231    Args:
232        data: Aggregated occupancy observations containing floor-space IDs,
233            occupancy states, original timestamps, and interval timestamps.
234
235    Returns:
236        A tuple containing:
237
238        - A DataFrame of eligible forecast rows, including the engineered
239          features and the matched future occupancy state in ``TARGET``.
240        - A dictionary of row counts describing the target-matching and
241          eligibility process.
242
243    Note:
244        The target is matched using the interval timestamp, not by selecting
245        the next available observation. Consequently, an observation is
246        excluded if the exact next five-minute bin is absent.
247
248        The observation-gap feature uses the original ``collecteddate``
249        timestamps. It is therefore possible for the elapsed time between
250        sensor observations to differ from five minutes, even when their
251        interval timestamps are consecutive.
252    """
253    # Work on a copy so feature engineering does not modify the aggregated
254    # observations used elsewhere in the evaluation.
255    data = data.copy()
256    data["current_state"] = data[TARGET]
257
258    # Calculate previous observations independently within each floor space
259    # to prevent readings from different spaces being mixed together.
260    grouped = data.groupby("floorspaceid", sort=False)
261    data["previous_state"] = grouped[TARGET].shift(1)
262    data["previous_timestamp"] = grouped["collecteddate"].shift(1)
263
264    # Measure the elapsed time between original sensor readings rather than
265    # assuming that consecutive interval bins are exactly five minutes apart.
266    data["gap_minutes"] = (
267        data["collecteddate"] - data["previous_timestamp"]
268    ).dt.total_seconds() / 60
269
270    # Exclude non-positive gaps and cap unusually large gaps before applying
271    # the logarithmic transform to reduce the influence of extreme values.
272    data.loc[data["gap_minutes"] <= 0, "gap_minutes"] = np.nan
273    data["log_gap_minutes"] = np.log1p(
274        data["gap_minutes"].clip(upper=1440)
275    )
276
277    # Extract calendar features from the current forecast-origin interval.
278    data["hour"] = data["interval"].dt.hour
279    data["day_of_week"] = data["interval"].dt.dayofweek
280    data["month"] = data["interval"].dt.month
281    data["is_weekend"] = (data["day_of_week"] >= 5).astype(int)
282
283    # Define the exact future interval that the current observation must predict.
284    data["target_interval"] = (
285        data["interval"] + pd.Timedelta(minutes=FORECAST_MINUTES)
286    )
287
288    # Build a lookup table of observed occupancy states indexed by their
289    # floor space and interval, allowing exact future-target matching.
290    future = data[["floorspaceid", "interval", TARGET]].rename(
291        columns={
292            "interval": "target_interval",
293            TARGET: "future_state",
294        }
295    )
296
297    # Match each forecast origin to the exact next interval.
298    # A left join preserves unmatched origins so they can be counted.
299    candidates = data.merge(
300        future,
301        on=["floorspaceid", "target_interval"],
302        how="left",
303        validate="one_to_one",
304        indicator=True,
305    )
306    total_rows = len(candidates)
307    missing_target_rows = int((candidates["_merge"] == "left_only").sum())
308
309    # Keep only rows with a matched future state and the historical
310    # information needed to calculate the forecasting features.
311    eligible = candidates.drop(columns="_merge").dropna(
312        subset=["future_state", "previous_state", "log_gap_minutes"]
313    ).copy()
314
315    # Store the matched future state as the prediction target and order
316    # the resulting records by target time and floor space.
317    eligible[TARGET] = eligible["future_state"]
318    eligible = eligible.sort_values(
319        ["target_interval", "floorspaceid"]
320    ).reset_index(drop=True)
321
322    # Track how many records are lost at each stage so forecast-row
323    # eligibility can be assessed independently of model performance.
324    counts = {
325        "aggregated_intervals": total_rows,
326        "rows_without_exact_next_bin": missing_target_rows,
327        "rows_with_exact_next_bin": total_rows - missing_target_rows,
328        "eligible_forecast_rows": len(eligible),
329        "rows_excluded_after_target_matching": (
330            total_rows - len(eligible)
331        ),
332    }
333    return eligible, counts

def create_windows(data: pandas.DataFrame) -> list[dict]: 
391def create_windows(data: pd.DataFrame) -> list[dict]:
392    """Generate three expanding chronological evaluation windows.
393
394    Unique target timestamps are sorted chronologically and divided using
395    cutoffs at 60%, 70%, 80%, and 90% of the available timestamps.
396
397    Each window uses all eligible observations up to the preceding cutoff
398    for training, followed by the next chronological segment for testing.
399    This produces three expanding training sets and three non-overlapping
400    test periods.
401
402    Args:
403        data: Eligible forecast rows containing a ``target_interval``
404            column.
405
406    Returns:
407        A list of three dictionaries. Each dictionary contains the window
408        name and its training-end, test-start, and test-end timestamps.
409
410    Raises:
411        ValueError: If fewer than 100 unique target timestamps are available
412            or if the cutoffs cannot produce non-empty chronological windows.
413
414    Note:
415        Approximately the first 60% of unique target timestamps are used
416        for the first training set. The three test periods cover approximately
417        60–70%, 70–80%, and 80–90% of the timestamps. The final 10% is not
418        included in these evaluation windows.
419    """
420    # Split using unique target timestamps rather than individual rows so
421    # observations sharing a target time cannot be divided across sets.
422    unique_times = np.sort(data["target_interval"].dropna().unique())
423    if len(unique_times) < 100:
424        raise ValueError(
425            "Not enough unique target timestamps for rolling evaluation."
426        )
427
428    n_times = len(unique_times)
429
430    # Establish chronological boundaries for three consecutive test periods.
431    # Each later training set includes the earlier training and test periods.
432    cutoffs = [int(n_times * fraction) for fraction in (0.60, 0.70, 0.80, 0.90)]
433    if any(cutoffs[i] >= cutoffs[i + 1] for i in range(len(cutoffs) - 1)):
434        raise ValueError("Not enough timestamps to create non-empty windows.")
435
436    # Use the preceding timestamp as the training boundary and the next
437    # segment as the test period for each expanding window.
438    return [
439        {
440            "window": f"Window {index + 1}",
441            "train_end": unique_times[cutoffs[index] - 1],
442            "test_start": unique_times[cutoffs[index]],
443            "test_end": unique_times[cutoffs[index + 1] - 1],
444        }
445        for index in range(3)
446    ]

def evaluate_windows( data: pandas.DataFrame, windows: list[dict]) -> tuple[pandas.DataFrame, pandas.DataFrame]: 
449def evaluate_windows(
450    data: pd.DataFrame,
451    windows: list[dict],
452) -> tuple[pd.DataFrame, pd.DataFrame]:
453    """Train and evaluate forecasting models across chronological windows.
454
455    For each window, the function selects training and test rows according
456    to their target timestamps. It then evaluates three prediction
457    approaches:
458
459    - Majority baseline: predicts the most common training target class.
460    - Persistence baseline: predicts that the current state continues
461      into the target interval.
462    - Random Forest: fits the configured classification pipeline using
463      the training data and predicts the test data.
464
465    Accuracy, balanced accuracy, macro F1, and weighted F1 are calculated
466    for each model and window. Individual predictions are also retained
467    for later inspection.
468
469    Args:
470        data: Eligible forecast rows containing the engineered features,
471            target labels, floor-space IDs, and target timestamps.
472        windows: Chronological window definitions generated by
473            ``create_windows``.
474
475    Returns:
476        A tuple containing:
477
478        - A DataFrame of evaluation metrics, with one row per model
479          and completed window.
480        - A DataFrame containing the actual and predicted states for
481          every evaluated test row.
482
483    Raises:
484        ValueError: If chronological overlap is detected between a
485            training and test set.
486        RuntimeError: If no evaluation windows complete successfully.
487
488    Note:
489        Windows with empty training or test sets, or fewer than two
490        training classes, are skipped. Training data expands between
491        windows, while the test periods remain separate.
492
493        The F1 metrics use ``STATE_ORDER`` as the fixed set of class
494        labels. This ensures that the reported macro and weighted F1
495        scores use a consistent class ordering across windows.
496    """
497    results = []
498    prediction_tables = []
499
500    for window in windows:
501        name = window["window"]
502        train_end = pd.Timestamp(window["train_end"])
503        test_start = pd.Timestamp(window["test_start"])
504        test_end = pd.Timestamp(window["test_end"])
505
506        # Select all targets up to the training cutoff and only the
507        # corresponding later chronological segment for testing.
508        train = data[data["target_interval"] <= train_end].copy()
509        test = data[
510            (data["target_interval"] >= test_start)
511            & (data["target_interval"] <= test_end)
512        ].copy()
513
514        # Skip windows that cannot support a meaningful model evaluation.
515        if train.empty or test.empty:
516            print(f"Skipping {name}: empty training or test set.")
517            continue
518        if train["target_interval"].max() >= test["target_interval"].min():
519            raise ValueError(f"Chronological overlap detected in {name}.")
520        if train[TARGET].nunique() < 2:
521            print(f"Skipping {name}: fewer than two training classes.")
522            continue
523
524        # Separate the model inputs from the future occupancy state to be
525        # predicted for each training and test observation.
526        X_train, y_train = train[FEATURES], train[TARGET]
527        X_test, y_test = test[FEATURES], test[TARGET]
528
529        # The majority baseline uses only the most frequent training class.
530        majority_class = y_train.value_counts().idxmax()
531        majority_predictions = np.repeat(majority_class, len(test))
532
533        # The persistence baseline assumes the current occupancy state
534        # remains unchanged over the forecast horizon.
535        persistence_predictions = X_test["current_state"].to_numpy()
536
537        # Fit a fresh Random Forest for each window so that each model
538        # uses only the training data available at that point in time.
539        model = make_model()
540        model.fit(X_train, y_train)
541        rf_predictions = model.predict(X_test)
542
543        # Keep all three prediction approaches together for consistent
544        # metric calculation and output generation.
545        predictions_by_model = {
546            "Majority baseline": majority_predictions,
547            "Persistence baseline": persistence_predictions,
548            "Random Forest": rf_predictions,
549        }
550
551        print("\n" + "-" * 70)
552        print(name)
553        print(f"Training targets through: {train_end}")
554        print(f"Testing targets: {test_start} to {test_end}")
555        print(f"Training rows: {len(train):,}")
556        print(f"Testing rows:  {len(test):,}")
557
558        # Evaluate every model against the same actual test labels.
559        for model_name, predictions in predictions_by_model.items():
560            results.append({
561                "window": name,
562                "model": model_name,
563                "train_rows": len(train),
564                "test_rows": len(test),
565                "train_end": train_end,
566                "test_start": test_start,
567                "test_end": test_end,
568                "accuracy": accuracy_score(y_test, predictions),
569                "balanced_accuracy": balanced_accuracy_score(y_test, predictions),
570                "macro_f1": f1_score(
571                    y_test, predictions, labels=STATE_ORDER,
572                    average="macro", zero_division=0,
573                ),
574                "weighted_f1": f1_score(
575                    y_test, predictions, labels=STATE_ORDER,
576                    average="weighted", zero_division=0,
577                ),
578            })
579
580        # Preserve row-level predictions so model errors and individual
581        # occupancy-state changes can be inspected after evaluation.
582        prediction_tables.append(pd.DataFrame({
583            "window": name,
584            "target_interval": test["target_interval"].to_numpy(),
585            "floorspaceid": test["floorspaceid"].to_numpy(),
586            "actual_state": y_test.to_numpy(),
587            "majority_prediction": majority_predictions,
588            "persistence_prediction": persistence_predictions,
589            "random_forest_prediction": rf_predictions,
590        }))
591
592    # Prevent empty output files from being generated if every window
593    # was skipped during the evaluation process.
594    if not results or not prediction_tables:
595        raise RuntimeError("No rolling evaluation windows completed.")
596
597    return (
598        pd.DataFrame(results),
599        pd.concat(prediction_tables, ignore_index=True),
600    )
