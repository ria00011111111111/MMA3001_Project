02_occupancy_forecast_visualisation

Train, evaluate and visualise next-interval occupancy forecasts.

The script:

  1. Loads occupancy observations from a compressed CSV dataset.
  2. Aggregates observations into five-minute intervals, retaining the latest observation for each floor-space and interval.
  3. Constructs forecasting samples using consecutive five-minute intervals.
  4. Trains and evaluates a Random Forest using a chronological train/test split.
  5. Compares the model against majority-class and persistence baselines.
  6. Exports predictions, model metrics and feature importance.
  7. Generates diagnostic visualisations and a text summary.

All timestamps are interpreted and stored in UTC. The prediction target is the recorded sensor occupancy state at the next interval; it is not an independently verified measurement of physical occupancy.

 107def load_and_aggregate_data():
 108    """Load and aggregate occupancy observations from the ZIP dataset.
 109
 110    Reads the CSV in chunks to limit memory usage. Invalid timestamps and
 111    records missing required fields are removed. Observations are assigned
 112    to five-minute intervals, and the latest observation is retained for
 113    each floor-space and interval combination.
 114
 115    Returns:
 116        pd.DataFrame: Aggregated observations containing floor-space ID,
 117            recorded occupancy state, original timestamp and interval.
 118
 119    Raises:
 120        FileNotFoundError: If the ZIP file or an internal CSV is unavailable.
 121        ValueError: If no valid occupancy records are loaded.
 122    """
 123    if not ZIP_FILE.is_file():
 124        raise FileNotFoundError(
 125            f"Occupancy ZIP file was not found: {ZIP_FILE}"
 126        )
 127
 128    print("\nLoading occupancy data from ZIP...")
 129
 130    chunks = []
 131    total_rows = 0
 132
 133    with ZipFile(ZIP_FILE) as archive:
 134
 135        # Identify CSV files contained within the compressed dataset.
 136        csv_files = [
 137            name
 138            for name in archive.namelist()
 139            if name.lower().endswith(".csv")
 140        ]
 141
 142        if not csv_files:
 143            raise FileNotFoundError(
 144                "No CSV file was found inside the occupancy ZIP."
 145            )
 146
 147        with archive.open(csv_files[0]) as file:
 148
 149            # Process the dataset in chunks to avoid loading all raw
 150            # observations into memory simultaneously.
 151            for chunk in pd.read_csv(
 152                file,
 153                usecols=[
 154                    "floorspaceid",
 155                    TARGET,
 156                    "collecteddate",
 157                ],
 158                chunksize=CHUNK_SIZE,
 159                low_memory=False,
 160            ):
 161                total_rows += len(chunk)
 162
 163                # Convert timestamps to UTC. Invalid timestamps are
 164                # converted to NaT and removed during cleaning.
 165                chunk["collecteddate"] = pd.to_datetime(
 166                    chunk["collecteddate"],
 167                    utc=True,
 168                    errors="coerce",
 169                )
 170
 171                # Remove records missing information required for forecasting.
 172                chunk = chunk.dropna(
 173                    subset=[
 174                        "floorspaceid",
 175                        TARGET,
 176                        "collecteddate",
 177                    ]
 178                )
 179
 180                # Assign each observation to its corresponding
 181                # five-minute interval.
 182                chunk["interval"] = (
 183                    chunk["collecteddate"].dt.floor("5min")
 184                )
 185
 186                chunks.append(
 187                    chunk[
 188                        [
 189                            "floorspaceid",
 190                            TARGET,
 191                            "collecteddate",
 192                            "interval",
 193                        ]
 194                    ]
 195                )
 196
 197                print(
 198                    f"Processed {total_rows:,} raw records..."
 199                )
 200
 201    if not chunks:
 202        raise ValueError(
 203            "No valid occupancy records were loaded."
 204        )
 205
 206    print("\nCombining and aggregating records...")
 207
 208    data = pd.concat(chunks, ignore_index=True)
 209    del chunks
 210
 211    # Sort observations chronologically within each floor-space and
 212    # interval so that the latest recorded observation can be retained.
 213    data = data.sort_values(
 214        [
 215            "floorspaceid",
 216            "interval",
 217            "collecteddate",
 218        ]
 219    )
 220
 221    # Retain only the latest observation for each floor-space and
 222    # five-minute interval combination.
 223    data = data.drop_duplicates(
 224        subset=[
 225            "floorspaceid",
 226            "interval",
 227        ],
 228        keep="last",
 229    )
 230
 231    # Restore chronological ordering for subsequent forecasting operations.
 232    data = data.sort_values(
 233        [
 234            "floorspaceid",
 235            "interval",
 236        ]
 237    ).reset_index(drop=True)
 238
 239    print(f"Raw records read: {total_rows:,}")
 240    print(f"Valid aggregated intervals: {len(data):,}")
 241    print(f"Floor spaces: {data['floorspaceid'].nunique()}")
 242    print(
 243        f"Interval range: {data['interval'].min()} "
 244        f"to {data['interval'].max()}"
 245    )
 246
 247    return data

 250def build_forecast_dataset(data):
 251    """Construct supervised samples for next-interval forecasting.
 252
 253    Each observation is paired with the following observation from the
 254    same floor-space. Only pairs exactly five minutes apart are retained.
 255    The current occupancy state and calendar variables are used as
 256    predictors, while the next interval's recorded state is the target.
 257
 258    Args:
 259        data (pd.DataFrame): Aggregated occupancy observations.
 260
 261    Returns:
 262        pd.DataFrame: Forecasting dataset containing predictor variables,
 263            forecast origins, target timestamps and future occupancy states.
 264
 265    Raises:
 266        ValueError: If no consecutive five-minute pairs are available.
 267    """
 268    print("\nConstructing next-interval forecasting targets...")
 269
 270    data = data.copy()
 271
 272    # Group by floor-space so that future observations are never
 273    # accidentally taken from a different floor-space.
 274    grouped = data.groupby(
 275        "floorspaceid",
 276        sort=False,
 277    )
 278
 279    # Shift each floor-space group independently to obtain its next
 280    # observation and corresponding timestamp.
 281    data["forecast_target_time"] = (
 282        grouped["interval"].shift(-1)
 283    )
 284
 285    data["actual_future_state"] = (
 286        grouped[TARGET].shift(-1)
 287    )
 288
 289    # Retain only pairs where the next observation occurs exactly
 290    # five minutes after the current forecast origin.
 291    expected_target = (
 292        data["interval"] + pd.Timedelta(minutes=5)
 293    )
 294
 295    data = data.loc[
 296        data["forecast_target_time"] == expected_target
 297    ].copy()
 298
 299    if data.empty:
 300        raise ValueError(
 301            "No consecutive five-minute interval pairs were found."
 302        )
 303
 304    # Define the current occupancy state and the future target as strings.
 305    data["current_state"] = data[TARGET].astype(str)
 306    data["actual_future_state"] = (
 307        data["actual_future_state"].astype(str)
 308    )
 309
 310    # Extract calendar-based predictors from the forecast-origin timestamp.
 311    data["hour"] = data["interval"].dt.hour
 312    data["day_of_week"] = data["interval"].dt.dayofweek
 313    data["month"] = data["interval"].dt.month
 314
 315    # Monday-Friday = 0; Saturday-Sunday = 1.
 316    data["is_weekend"] = (
 317        data["day_of_week"] >= 5
 318    ).astype(int)
 319
 320    print(
 321        f"Consecutive forecast pairs: {len(data):,}"
 322    )
 323
 324    return data

 331def evaluate_model(name, actual, predicted, labels):
 332    """Calculate classification metrics for a forecasting model.
 333
 334    Computes accuracy, balanced accuracy, macro F1 and weighted F1.
 335    Explicit class labels ensure that the F1 calculations use a consistent
 336    set of occupancy classes across the evaluated models.
 337
 338    Args:
 339        name (str): Model or baseline name.
 340        actual: Ground-truth occupancy states.
 341        predicted: Predicted occupancy states.
 342        labels (list): Occupancy classes included in the evaluation.
 343
 344    Returns:
 345        dict: Model name and calculated evaluation metrics.
 346    """
 347    return {
 348        "model": name,
 349
 350        "accuracy": accuracy_score(
 351            actual,
 352            predicted,
 353        ),
 354
 355        "balanced_accuracy": balanced_accuracy_score(
 356            actual,
 357            predicted,
 358        ),
 359
 360        "macro_f1": f1_score(
 361            actual,
 362            predicted,
 363            labels=labels,
 364            average="macro",
 365            zero_division=0,
 366        ),
 367
 368        "weighted_f1": f1_score(
 369            actual,
 370            predicted,
 371            labels=labels,
 372            average="weighted",
 373            zero_division=0,
 374        ),
 375    }

 378def generate_forecast_outputs(data):
 379    """Train and evaluate the Random Forest forecasting model.
 380
 381    Creates a chronological training and testing split using forecast
 382    origin times. Training targets are restricted to timestamps before
 383    the test period to prevent temporal overlap. The Random Forest is
 384    compared against majority-class and persistence baselines.
 385
 386    The function exports:
 387        - Held-out forecast predictions.
 388        - Model comparison metrics.
 389        - Random Forest feature importance.
 390
 391    Args:
 392        data (pd.DataFrame): Supervised forecasting dataset.
 393
 394    Returns:
 395        tuple: Prediction output, model comparison DataFrame and feature
 396            importance DataFrame.
 397
 398    Raises:
 399        ValueError: If the dataset cannot produce valid training and test sets.
 400        AssertionError: If temporal overlap is detected at the split boundary.
 401    """
 402
 403    # Separate categorical identifiers and states from numerical
 404    # calendar-based predictors.
 405    categorical_features = [
 406        "floorspaceid",
 407        "current_state",
 408    ]
 409
 410    numeric_features = [
 411        "hour",
 412        "day_of_week",
 413        "month",
 414        "is_weekend",
 415    ]
 416
 417    feature_columns = (
 418        categorical_features + numeric_features
 419    )
 420
 421    X = data[feature_columns].copy()
 422    y = data["actual_future_state"].copy()
 423
 424    # Use unique forecast-origin timestamps to establish a chronological
 425    # split rather than randomly splitting individual observations.
 426    unique_times = np.sort(
 427        data["interval"].unique()
 428    )
 429
 430    if len(unique_times) < 2:
 431        raise ValueError(
 432            "Not enough distinct forecast-origin times to split."
 433        )
 434
 435    split_index = int(
 436        len(unique_times) * (1 - TEST_FRACTION)
 437    )
 438
 439    # Ensure that both training and testing periods contain timestamps.
 440    split_index = min(
 441        max(split_index, 1),
 442        len(unique_times) - 1,
 443    )
 444
 445    cutoff_time = unique_times[split_index]
 446
 447    # Training targets must also finish before the test forecast-origin
 448    # period. This prevents temporal overlap across the split boundary.
 449    train_mask = (
 450        (data["interval"] < cutoff_time)
 451        & (data["forecast_target_time"] < cutoff_time)
 452    )
 453
 454    test_mask = (
 455        data["interval"] >= cutoff_time
 456    )
 457
 458    X_train = X.loc[train_mask]
 459    X_test = X.loc[test_mask]
 460
 461    y_train = y.loc[train_mask]
 462    y_test = y.loc[test_mask]
 463
 464    if X_train.empty or X_test.empty:
 465        raise ValueError(
 466            "Chronological split produced an empty train or test set."
 467        )
 468
 469    if y_train.nunique() < 2:
 470        raise ValueError(
 471            "Training data contains fewer than two target classes."
 472        )
 473
 474    # Independently check that the final training target occurs before
 475    # the first testing forecast origin.
 476    train_target_end = (
 477        data.loc[train_mask, "forecast_target_time"].max()
 478    )
 479
 480    test_origin_start = (
 481        data.loc[test_mask, "interval"].min()
 482    )
 483
 484    if train_target_end >= test_origin_start:
 485        raise AssertionError(
 486            "Temporal leakage detected across the train/test boundary."
 487        )
 488
 489    print("\nChronological split")
 490    print(f"Training forecast pairs: {len(X_train):,}")
 491    print(f"Testing forecast pairs: {len(X_test):,}")
 492    print(f"Training target period ends: {train_target_end}")
 493    print(f"Testing forecast origins begin: {test_origin_start}")
 494
 495    # Majority-class baseline: always predict the most frequent
 496    # future occupancy state in the training dataset.
 497    majority_state = y_train.value_counts().idxmax()
 498
 499    majority_predictions = np.full(
 500        len(y_test),
 501        majority_state,
 502        dtype=object,
 503    )
 504
 505    # Persistence baseline: assume the next occupancy state remains
 506    # unchanged from the current recorded state.
 507    persistence_predictions = (
 508        data.loc[test_mask, "current_state"].to_numpy()
 509    )
 510
 511    # One-hot encode categorical predictors and pass numerical
 512    # calendar features through without additional transformation.
 513    preprocessor = ColumnTransformer(
 514        transformers=[
 515            (
 516                "categorical",
 517                OneHotEncoder(handle_unknown="ignore"),
 518                categorical_features,
 519            ),
 520            (
 521                "numeric",
 522                "passthrough",
 523                numeric_features,
 524            ),
 525        ]
 526    )
 527
 528    # Use a pipeline so preprocessing is fitted using training data only.
 529    # This ensures that test observations do not influence preprocessing.
 530    model = Pipeline(
 531        steps=[
 532            (
 533                "preprocessor",
 534                preprocessor,
 535            ),
 536            (
 537                "classifier",
 538                RandomForestClassifier(
 539                    n_estimators=100,
 540                    max_depth=12,
 541                    min_samples_leaf=10,
 542                    class_weight="balanced_subsample",
 543                    random_state=RANDOM_STATE,
 544                    n_jobs=-1,
 545                ),
 546            ),
 547        ]
 548    )
 549
 550    print("\nTraining Random Forest...")
 551
 552    model.fit(
 553        X_train,
 554        y_train,
 555    )
 556
 557    # Generate predictions for the chronologically held-out test period.
 558    rf_predictions = model.predict(X_test)
 559
 560    # Use a consistent set of class labels when calculating metrics.
 561    labels = sorted(
 562        set(y_train.unique()) | set(y_test.unique())
 563    )
 564
 565    # Compare the Random Forest against both baseline approaches.
 566    comparison = pd.DataFrame([
 567        evaluate_model(
 568            "Majority-class baseline",
 569            y_test,
 570            majority_predictions,
 571            labels,
 572        ),
 573        evaluate_model(
 574            "Persistence",
 575            y_test,
 576            persistence_predictions,
 577            labels,
 578        ),
 579        evaluate_model(
 580            "Random Forest",
 581            y_test,
 582            rf_predictions,
 583            labels,
 584        ),
 585    ])
 586
 587    comparison.to_csv(
 588        COMPARISON_FILE,
 589        index=False,
 590    )
 591
 592    # Retrieve the transformed feature names and their corresponding
 593    # Random Forest impurity-based importance values.
 594    feature_names = (
 595        model.named_steps["preprocessor"].get_feature_names_out()
 596    )
 597
 598    importances = (
 599        model.named_steps["classifier"].feature_importances_
 600    )
 601
 602    importance = pd.DataFrame({
 603        "feature": feature_names,
 604        "importance": importances,
 605    }).sort_values(
 606        "importance",
 607        ascending=False,
 608    )
 609
 610    importance.to_csv(
 611        IMPORTANCE_FILE,
 612        index=False,
 613    )
 614
 615    # Assemble the held-out predictions with their forecast origins,
 616    # target timestamps and actual future states.
 617    prediction_output = data.loc[
 618        test_mask,
 619        [
 620            "interval",
 621            "forecast_target_time",
 622            "actual_future_state",
 623            "floorspaceid",
 624        ],
 625    ].copy()
 626
 627    prediction_output = prediction_output.rename(
 628        columns={
 629            "interval": "forecast_origin",
 630        }
 631    )
 632
 633    prediction_output["random_forest_prediction"] = (
 634        rf_predictions
 635    )
 636
 637    prediction_output["persistence_prediction"] = (
 638        persistence_predictions
 639    )
 640
 641    prediction_output = prediction_output[
 642        PREDICTION_COLUMNS
 643    ]
 644
 645    prediction_output.to_csv(
 646        PREDICTIONS_FILE,
 647        index=False,
 648    )
 649
 650    print("\nFORECAST MODEL COMPARISON")
 651    print(
 652        comparison.to_string(
 653            index=False,
 654            float_format="%.4f",
 655        )
 656    )
 657
 658    print("\nForecast CSV files saved:")
 659    print(PREDICTIONS_FILE)
 660    print(COMPARISON_FILE)
 661    print(IMPORTANCE_FILE)
 662
 663    return (
 664        prediction_output,
 665        comparison,
 666        importance,
 667    )

1176def calculate_class_f1(actual, predicted, state):
1177    """Calculate the one-vs-rest F1 score for an occupancy state.
1178
1179    Treats the selected state as the positive class and all other states
1180    as the negative class.
1181
1182    Args:
1183        actual: Ground-truth occupancy states.
1184        predicted: Predicted occupancy states.
1185        state (str): Occupancy state being evaluated.
1186
1187    Returns:
1188        float: F1 score for the selected occupancy state.
1189    """
1190
1191    actual = np.asarray(actual)
1192    predicted = np.asarray(predicted)
1193
1194    # Calculate true positives, false positives and false negatives
1195    # for the selected occupancy state.
1196    tp = np.sum(
1197        (actual == state) & (predicted == state)
1198    )
1199
1200    fp = np.sum(
1201        (actual != state) & (predicted == state)
1202    )
1203
1204    fn = np.sum(
1205        (actual == state) & (predicted != state)
1206    )
1207
1208    denominator = (
1209        2 * tp + fp + fn
1210    )
1211
1212    # Return zero if the class has no positive predictions or observations.
1213    return (
1214        2 * tp / denominator
1215        if denominator
1216        else 0.0
1217    )
